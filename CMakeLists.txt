# Optional SoapySDR module (driver=hydra). Enabled when SoapySDR is installed;
# the core library and hydra_rx never depend on it. project() is C-only, so
# the module's C++ sources need the CXX language enabled explicitly.
option(HYDRASDR_SOAPY "Build the SoapySDR module when SoapySDR is available" ON)
if(HYDRASDR_SOAPY)
    find_package(SoapySDR CONFIG QUIET)
    if(SoapySDR_FOUND)
        enable_language(CXX)
        add_subdirectory(soapy)
    else()
        message(STATUS "SoapySDR not found; skipping the hydra module")
    endif()
endif()
